# ShieldGemma-2B - Task 1 benchmark dependencies.
#
# Self-contained: install into a dedicated venv for this experiment only,
# so it cannot collide with the repo-wide requirements.txt (see
# docs/task1-shieldhgemma.md).
#
#   python3 -m venv .venv-shieldgemma
#   source .venv-shieldgemma/bin/activate
#   pip install -r src/ShieldGemma/requirements.txt
#
# GPU note: on a CUDA box install the matching torch wheel FIRST, then
# this file:
#   pip install torch --index-url https://download.pytorch.org/whl/cu121
#   pip install -r src/ShieldGemma/requirements.txt

# ---------------- Core ----------------
# ShieldGemma is inference-only here: one forward pass per (input,
# policy) pair, no training, no generation.
torch>=2.1
numpy>=1.23
tqdm>=4.60

# ---------------- HuggingFace stack ----------------
# Upper bound is deliberate. transformers 5.x requires torch >= 2.4,
# which is unavailable on macOS x86_64 (PyTorch stopped shipping those
# wheels after 2.2.2) - installing 5.x there silently disables the
# PyTorch backend and every model load fails with
# "AutoModelForCausalLM requires the PyTorch library".
# Lower bound 4.42 is the first release with the Gemma-2 architecture
# that ShieldGemma-2B is built on. 4.57.1 is the verified combination.
transformers>=4.42,<5
accelerate>=0.26
huggingface_hub>=0.23        # snapshot_download(local_dir=...) writes real files
safetensors>=0.4.2

# ---------------- Quantisation (optional) ----------------
# Only needed for `--dtype int8` / `--dtype nf4`. CUDA only; macOS users
# stay on bfloat16 / float32.
bitsandbytes>=0.43; platform_system != "Darwin"
