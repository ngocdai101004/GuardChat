# Llama-Guard-3-8B - Task 1 benchmark dependencies.
#
# Self-contained: install into a dedicated venv for this experiment only,
# so it cannot collide with the repo-wide requirements.txt (see
# docs/task1-llamaguard.md).
#
#   python3 -m venv .venv-llamaguard
#   source .venv-llamaguard/bin/activate
#   pip install -r src/LlamaGuard/requirements.txt
#
# GPU note: on a CUDA box install the matching torch wheel FIRST, then
# this file:
#   pip install torch --index-url https://download.pytorch.org/whl/cu121
#   pip install -r src/LlamaGuard/requirements.txt

# ---------------- Core ----------------
# Llama-Guard is inference-only here: a ~20-token greedy generation per
# input, no training.
torch>=2.1
numpy>=1.23
tqdm>=4.60

# ---------------- HuggingFace stack ----------------
# Lower bound 4.43 is the first release with Llama 3.1 support, which
# Llama-Guard-3-8B is built on; it also carries the chat template that
# accepts a `categories=` override for the custom GuardChat taxonomy.
# Upper bound is deliberate: transformers 5.x requires torch >= 2.4,
# unavailable on macOS x86_64 (PyTorch stopped shipping those wheels
# after 2.2.2) - installing 5.x there silently disables the PyTorch
# backend and every model load fails with
# "AutoModelForCausalLM requires the PyTorch library".
transformers>=4.43,<5
accelerate>=0.26
huggingface_hub>=0.23        # snapshot_download(local_dir=...) writes real files
safetensors>=0.4.2

# ---------------- Quantisation (optional) ----------------
# Needed for `--dtype int8` / `--dtype nf4`, which is how the 8B model
# fits on a <16 GB GPU. CUDA only; macOS users must stay on bfloat16 or
# float32 and have ~16-32 GB of RAM free.
bitsandbytes>=0.43; platform_system != "Darwin"
